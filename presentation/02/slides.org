* Extend DataType with records

- Extend the type system:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  inductive DataType where
    | text
    | boolean
    | natural
    ..
    | record (fieldNames : Enum) (fieldTypes : fieldNames → DataType)
  #+end_src
\pause{}
- Extend the interpretation:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  abbrev DataType.denote (type : DataType) : Type :=
    match type with
    | .text => String
    | .boolean => Bool
    | .natural => Nat
    ..
    | .record fieldNames fieldTypes =>
       (i : fieldNames) → DataType.denote (fieldTypes i)
  #+end_src

* Notation
- We write ⟦t⟧ for the denotation/interpretation of =t= in Lean. \pause{}
- In Lean:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  scoped notation "⟦" t "⟧" => DataType.denote t
  #+end_src
- Examples:
  1. ⟦.natural⟧ = Nat
  2. ⟦.boolean⟧ = Bool
  3. ⟦.natural⟧ → ⟦.boolean⟧ = Nat → Bool

* UI components
+=Form=+ has been replaced by *=Component=*.
A =Component= describes a UI over a state.

#+attr_latex: :options basicstyle=\ttfamily\footnotesize,deletekeywords={[3]{value}}
#+begin_src lean
inductive Component : (state : DataType) → Type 1 where
  | field
      {state value : DataType}
      (path : Path state value)
      (label : String)
      (widget : Widget value)
      : Component state
#+end_src
\pause{}
- How to have multiple components?

* Combine components
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
inductive Component : (state : DataType) → Type 1 where
  | field ..
  | stack
      {state : DataType}
      {items : Enum}
      (label : Option String := none)
      (children : items → Component state)
      : Component state
#+end_src

* A component for the address record
Each field has a path selecting which part of the state it edits.

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def addressForm : Component address :=
  .stack (some "Address") (items := addressFields) (children := fun
    | .street => Component.field (Path.field .street .self) "Street" .textInput
    | .number => Component.field (Path.field .number .self) "Number" .naturalInput
    | .kind => Component.field (Path.field .kind .self) "Kind" .select)
#+end_src

* Typed paths
A path is a typed route from a type to a type within it:
#+attr_latex: :options basicstyle=\ttfamily\footnotesize,aboveskip=5pt,belowskip=5pt
#+begin_src lean
inductive Path : (source : DataType) → (target : DataType) → Type 1 where
  | self {type : DataType} : Path type type
  | field
      {fieldNames : Enum}
      {fieldTypes : fieldNames → DataType}
      {target : DataType}
      (fieldName : fieldNames)
      (rest : Path (fieldTypes fieldName) target)
      : Path (.record fieldNames fieldTypes) target
#+end_src

* =Path= operations
- Get
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=5pt,belowskip=5pt,deletekeywords={[3]{value}}
  #+begin_src lean
  def Path.get {source target : DataType}
      (path : Path source target)
      (value : ⟦source⟧)
      : ⟦target⟧ := ..
  #+end_src
\pause{}
- Set
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=5pt,belowskip=5pt,deletekeywords={[3]{value}}
  #+begin_src lean
  def Path.set {source target : DataType} (path : Path source target)
      (replacement : ⟦target⟧)
      (value : ⟦source⟧)
      : ⟦source⟧ := ..
  #+end_src
\pause{}
- Compose
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=5pt,belowskip=5pt
  #+begin_src lean
  def Path.trans {a b c : DataType}
      (p1 : Path a b)
      (p2 : Path b c)
      : Path a c := ..
  #+end_src

* Address components: without and with =group=
- Can we simplify building components for records? \pause{}
- Using =stack= directly:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=3pt,belowskip=3pt
  #+begin_src lean
  def addressForm : Component address :=
    .stack (some "Address") (items := addressFields) (children := fun
      | .street => Component.field (Path.field .street .self) "Street" .textInput
      | .number => Component.field (Path.field .number .self) "Number" .naturalInput
      | .kind => Component.field (Path.field .kind .self) "Kind" .select)
  #+end_src
\pause{}
- With =group= helper:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=3pt,belowskip=3pt
  #+begin_src lean
  def addressFormWithGroup : Component address :=
    Builder.group (fields := addressFields) "Address" (children := fun
      | .street => Builder.field "Street" .textInput
      | .number => Builder.field "Number" .naturalInput
      | .kind => Builder.field "Kind" .select)
    |>.build
  #+end_src

* How =group= works
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
abbrev Builder (state value : DataType) :=
  Path state value → Component state

def Builder.build {state : DataType}
    (builder : Builder state state) : Component state :=
  builder .self

def Builder.group
    {state : DataType} {fields : Enum} {types : fields → DataType}
    (label : String)
    (children : (i : fields) → Builder state (types i))
    (displayOrder : Equiv.Perm fields := Equiv.refl _)
    : Builder state (.record fields types) :=
  fun path =>
    .stack (some label)
      (children := fun i => children i (path.trans (.field i .self)))
      displayOrder
#+end_src
\pause{}
- By using =group= we get *convenient syntax* and we guarantee that *each field is
  displayed exactly once*.

* Typed expressions
We want typed expressions that read a value and compute a result.
#+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=4pt,belowskip=4pt
#+begin_src lean
inductive Expr (input : DataType) : (result : DataType) → Type 1 where
  ..
#+end_src
1. Compare natural numbers and dates. \pause{}
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=4pt,belowskip=4pt
  #+begin_src lean
  | natLe (l r : Expr input .natural) : Expr input .boolean
  | dateLe (l r : Expr input .date) : Expr input .boolean
  #+end_src
2. Compute boolean conjunction. \pause{}
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=4pt,belowskip=4pt
  #+begin_src lean
  | and (l r : Expr input .boolean) : Expr input .boolean
  #+end_src
3. Project a value out of a record. \pause{}
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize,aboveskip=4pt,belowskip=4pt
  #+begin_src lean
  | project {t : DataType} (p : Path input t) : Expr input t
  #+end_src

* Typed expressions
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
inductive Expr (input : DataType) : (result : DataType) → Type 1 where
  | value {t : DataType} (v : ⟦t⟧) : Expr input t
  | dateLe (l r : Expr input .date) : Expr input .boolean
  | natLe (l r : Expr input .natural) : Expr input .boolean
  | and (l r : Expr input .boolean) : Expr input .boolean
  | project {t : DataType} (p : Path input t) : Expr input t
#+end_src

- We’ll add constructors to =Expr= as we need more operations. \pause{}

* Extending the language: sums
Let's add *sum types* to the language. \pause{}

A sum type can be seen as an extension of an enum. Each of the cases can have a
payload. E.g.

#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
inductive Shape where
  | dot
  | circle (radius : Nat)
  | rectangle (width height : Nat)
#+end_src
\pause{}

- How to encode sum types in our language?

* Extend =DataType= with sums
- Extend the type system:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  inductive DataType where
    ..
    | sum (tags : Enum) (constructors : tags → DataType)
  #+end_src
\pause{}
- Extend the interpretation:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  structure SumValue {Tag : Type} (Payload : Tag → Type) where
    tag : Tag
    payload : Payload tag

  abbrev DataType.denote (t : DataType) : Type :=
    match t with
    ..
    | .sum _ constructors =>
        SumValue (fun tag => DataType.denote (constructors tag))
  #+end_src

* Name the shape tags
The constructor names are defined with an =Enum=:
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
inductive ShapeTag where
  | dot
  | circle
  | rectangle

instance : FinEnum ShapeTag := ..

def shapeTags : Enum where
  type := ShapeTag
  label
    | .dot => "dot"
    | .circle => "circle"
    | .rectangle => "rectangle"
#+end_src
- Terminology: Constructor name = Case name = Variant name = Tag.

* Example: shapes

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
abbrev unit : DataType := .record noFields Empty.elim

abbrev rectangle : DataType := .record rectangleFields fun
  | .width => .natural
  | .height => .natural

abbrev constructors : shapeTags → DataType
  | .dot => unit
  | .circle => .natural
  | .rectangle => rectangle

abbrev shape : DataType := .sum shapeTags constructors

def initial : ⟦shape⟧ := ⟨.circle, 3⟩
#+end_src

* Extend =Component= with sums
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
inductive Component : (state : DataType) → Type 1 where
  ..
  | sum
      {state : DataType}
      {tags : Enum}
      {constructors : tags → DataType}
      (path : Path state (.sum tags constructors))
      (label : String)
      (branches : (i : tags) → Component (constructors i))
      (defaults : (i : tags) → ⟦constructors i⟧)
      : Component state
#+end_src

* A form for shapes
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def form : Component shape :=
  .sum .self "Shape"
    (branches := fun
      | .dot => .stack none (items := noFields) (children := fun i => nomatch i)
      | .circle => .field .self "Radius" .naturalInput
      | .rectangle => .stack none (items := rectangleFields) (children := fun
          | .width => .field (.field .width .self) "Width" .naturalInput
          | .height => .field (.field .height .self) "Height" .naturalInput))
    (defaults := fun
      | .dot => fun i => nomatch i
      | .circle => 3
      | .rectangle => fun
        | .width => 4
        | .height => 2)
#+end_src
