* Source and status
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive HttpMethod where
  | get
  | post

structure HttpRequest where
  method : HttpMethod
  url : String
  body : Option String

inductive Source (α : Type) where
  | pure (value : α)
  | fetch (request : HttpRequest) (decode : Json → Except String α)

inductive Status (α : Type) where
  | loading
  | ready (value : α)
  | failed (message : String)
#+end_src

* Load: the primitive
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive ComponentNode .. where
  ..
  | load {Γ : Context}
      {α β data : Type}
      {states : Status data → Type}
      {holes : Nat}
      [inst : DecidableEq data]
      (source : Computation Γ α (Source data))
      (body : (status : Status data) → ComponentNode Γ α (states status) β holes)
      : ComponentNode Γ α (LoadedState data states) β holes

structure LoadedState (data : Type) (state : Status data → Type) where
  status : Status data
  body : state status
#+end_src

- Each status has its own state type. E.g. while loading a table: nothing to
  remember; once loaded: which row is selected.

* loadWith: the common helper
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
def Component.loadWith
  {Γ : Context}
  {α data : Type}
  [inst : DecidableEq data]
  (source : Computation Γ α (Source data))
  (body : data → Component Γ α)
  : Component Γ α :=
  .load source fun
    | .ready data => body data
    | .loading => .computed "" (.const "Loading…")
    | .failed message => .computed "" (.const s!"Could not load: {message}")
#+end_src

* Typed endpoints
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
structure Api.Signature where
  Input : Type
  Output : Type
  [inputToJson : ToJson Input]
  [inputFromJson : FromJson Input]
  [outputToJson : ToJson Output]
  [outputFromJson : FromJson Output]

class Api (endpoint : Type) [FinEnum endpoint] where
  signature : endpoint → Api.Signature
  path : endpoint → String
  distinct : FinEnum.toList endpoint |>.map path |>.Nodup

Api.source
  (named : endpoint)
  (input : (Api.signature named).Input)
  : Source (Api.signature named).Output
#+end_src

* An API, by hand and with syntax
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive RatesEndpoint where
  | currencies
  | rate
  deriving DecidableEq, FinEnum

instance : Api RatesEndpoint where
  signature
    | .currencies => { Input := Unit, Output := List Currency.Raw }
    | .rate => { Input := Currency.Raw, Output := Nat }
  path
    | .currencies => "/api/currencies"
    | .rate => "/api/rate"
  distinct := by decide
#+end_src

* Lookup types
#+attr_latex: :options basicstyle=\ttfamily\footnotesize
#+begin_src lean
structure Reference (Key : Type) where
  id : Key
  displayText : String

structure LookupCandidate (Key Details : Type) where
  reference : Reference Key
  details : Details -- For later

structure LookupResults (Key Details : Type) where
  found : List (LookupCandidate Key Details)
  more : Bool -- true when found does not include all matches
  keys_nodup : (found.map LookupCandidate.wireKey).Nodup
#+end_src

inductive LookupAvailability where

Component.lookup

  : Component Γ α

- =search=: from the search text to a request.

* The delivery form
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
structure Delivery where
  country : Option (Reference String)
  city : Option (Reference String)
  deriving FormModel

endpoints DeliveryEndpoint where
  | countries : String → LookupResults String Unit at "/api/countries"
  | cities : CitySearch → LookupResults String Unit at "/api/cities"

.lookup path[Delivery.Field.city] "City"
  (compute%
    (country : Option (Reference String)) ← path[Delivery.Field.country]
    => fun text => match country with
      | some country => Api.source DeliveryEndpoint.cities { country := country.id, text }
      | none => .pure .empty)
  (.const LookupAvailability.always)
#+end_src

* Summary
- =load=: load data via http requests.
- =endpoints=: typed requests shared by client and server.
- =lookup=: search on the server.
