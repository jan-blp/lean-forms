* RefinedDataType: first design
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
structure ErrorLocation (root : DataType) where
  {t : DataType}
  path : Path root t

structure Constraint (root : DataType) where
  condition : Expr root .boolean
  errorMessage : String
  locations : List (ErrorLocation root)

inductive RefinedDataType : (unrefined : DataType) → Type 1 where
  | base (t : DataType) : RefinedDataType t
  | refine {t : DataType}
      (refined : RefinedDataType t)
      (constraint : Constraint t)
      : RefinedDataType t
  | sum .. -- recurse into sum constructors
  | record .. -- recurse into record fields
  | list .. -- recurse into list elements
#+end_src

* Denotation of refined types
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def atLeastFive : Expr .natural .boolean :=
  .natLe (.value 5) (.project .self)

def atLeastFiveType : RefinedDataType .natural :=
  .refine (.base .natural)
    { condition := atLeastFive
      errorMessage := "Must be at least 5."
      locations := [⟨.self⟩] }
#+end_src

We want the denotation to be:

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
structure _ where
  val : Nat
  isValid : atLeastFive.eval val = true
#+end_src

* Denotation of refined types
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def RefinedDataType.denotation (refined : RefinedDataType t)
    : RefinedDataType.Denotation t :=
  match refined with
  | .base t => ⟨⟦t⟧, id⟩
  | .refine refined constraint =>
      let inner := refined.denotation
      { ValidatedValue :=
        { value : inner.ValidatedValue //
          constraint.condition.eval (inner.erase value) = true }
        erase := fun value => inner.erase value.val }
  -- record, list, sum: preserve their structure
#+end_src

* Typed validation
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def RefinedDataType.validate
  (refined : RefinedDataType t)
  (value : ⟦t⟧)
  : Option refined.ValidatedValue :=
  match refined with
  | .base _ => some value
  | .refine inner constraint => do
      let validated ← inner.validate value
      if h : constraint.condition.eval (inner.erase validated) = true then
        return ⟨validated, h⟩
      else none
  -- structural cases have the expected implementation
#+end_src

* Checking widths
Checks that every shape fits within the canvas:
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def widthsFit : Expr drawing .boolean :=
  .all (.project (.field .shapes .self))
    (.natLe
      (.cases (.project (.field .item .self)) fun
        | .circle => .natMul (.value 2) (.project .self)
        | .rectangle => .project (.field .width .self))
      (.project (.field .outer (.field DrawingField.side .self))))

def widthConstraint : Constraint drawing where
  condition := widthsFit
  errorMessage := "A shape is wider than the canvas."
  locations := [⟨.field .shapes .self⟩]
#+end_src

- *Limitation*: We can't point to the offending element!

* Second design
Second design:
- Error locations must be more general than =Path=. I.e. they must be able to
  point to particular elements of a list and inside constructor payloads.
- Error Locations should be generated dynamically.

* Error locations
Similar to =Path=, but can also point to list elements and sum payloads.
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive Location : (root : DataType) → Type 1 where
  | self {root : DataType} : Location root -- same as Path
  | field
      {fieldNames : Enum} {fieldTypes : fieldNames → DataType}
      (fieldName : fieldNames)
      (rest : Location (fieldTypes fieldName))
      : Location (.record fieldNames fieldTypes) -- same as Path
  | item {element : DataType}
      (index : Nat)
      (rest : Location element)
      : Location (.list element) -- New
  | constructor
      {tags : Enum} {constructors : tags → DataType}
      (tag : tags)
      (rest : Location (constructors tag))
      : Location (.sum tags constructors) -- New
#+end_src
\pause{}
- Is it type safe?

* Computing error locations
Because locations are generated dynamically. At compile time we still don't know
the index of the offending element, so we need an expression that computes the
index:

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive LocationExpr (input : DataType) : (root : DataType) → Type 1 where
  | self {root : DataType} : LocationExpr input root
  | field
      {fieldNames : Enum} {fieldTypes : fieldNames → DataType}
      (fieldName : fieldNames)
      (rest : LocationExpr input (fieldTypes fieldName))
      : LocationExpr input (.record fieldNames fieldTypes)
  | item {element : DataType}
      (index : Expr input .natural) -- only difference
      (rest : LocationExpr input element)
      : LocationExpr input (.list element)
  | constructor
      {tags : Enum} {constructors : tags → DataType}
      (tag : tags)
      (rest : LocationExpr input (constructors tag))
      : LocationExpr input (.sum tags constructors)
#+end_src

* A checker
A =Checker= is a program that validates a value and produces error messages when
invalid.
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
Checker.run : {ty : DataType} → Checker ty → ⟦ty⟧ → List (ResolvedError ty)

structure ResolvedError (root : DataType) where
  message : String
  targets : TargetResolution root

inductive TargetResolution (root : DataType) where
  | global
  | locations (locations : List (Location root))
  | invalidTarget (location : Location root) -- BUG
#+end_src

* What can a checker do?
1. Report an error: =.report message targets=
2. Sequence two checkers: =.seq first second=
3. Choose a branch: =.ite condition thenBranch elseBranch=
4. Iterate a list: =.each items body=
5. Match a sum: =.cases subject branches=
6. Do nothing: =.pass=

* Checker definition
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive ScopedChecker (root : DataType) : (input : DataType) → Type 1 where
  | pass {input : DataType} : ScopedChecker root input
  | report {input : DataType}
      (message : Expr input .text) (targets : TargetsExpr root input)
      : ScopedChecker root input
  | ite {input : DataType}
      (condition : Expr input .boolean)
      (thenBranch elseBranch : ScopedChecker root input)
      : ScopedChecker root input
  | seq {input : DataType} (left right : ScopedChecker root input)
      : ScopedChecker root input
  | each {input element : DataType}
      (items : Expr input (.list element))
      (body : ScopedChecker root (IterationContext input element))
      : ScopedChecker root input
  | cases {input : DataType} {tags : Enum} {constructors : tags → DataType}
      (subject : Expr input (.sum tags constructors))
      (branches : (tag : tags) →
        ScopedChecker root (BranchContext input (constructors tag)))
      : ScopedChecker root input

abbrev Checker (ty : DataType) := ScopedChecker ty ty
#+end_src

* A checker: example
For each shape, check that its diameter or width fits the canvas. If it is too
wide, report an error at that shape's radius or width. Note that =.when= is a
helper defined in terms of =ite= and =pass=.

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
def widthChecker : Checker drawing :=
  .each (.project (.field .shapes .self))
    (.cases (.project (.field .item .self)) fun
      | .circle =>
        .when (.not (.natLe (.natMul (.value 2) (.project (.field .payload .self)))
          (.project (.field .outer (.field .outer (.field .side .self))))))
          (.report (.value "Circle is wider than the canvas.")
            (.locations [.field .shapes
              (.item (.project (.field .outer (.field .index .self)))
                (.constructor .circle .self))]))
      | .rectangle =>
        .when (.not (.natLe (.project (.field .payload (.field .width .self)))
          (.project (.field .outer (.field .outer (.field .side .self))))))
          (.report (.value "Rectangle is wider than the canvas.")
            (.locations [.field .shapes
              (.item (.project (.field .outer (.field .index .self)))
                (.constructor .rectangle (.field .width .self)))])))
#+end_src

* Specification of a checker
How can we ensure a checker does what we want?
1. We write a specification:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  def WidthsFit (value : ⟦drawing⟧) : Prop :=
    ∀ s ∈ value .shapes, match s with
      | ⟨.circle, radius⟩ => 2 * radius ≤ value .side
      | ⟨.rectangle, sides⟩ => sides .width ≤ value .side
  #+end_src
2. And we prove soundness:
  #+attr_latex: :options basicstyle=\ttfamily\scriptsize
  #+begin_src lean
  theorem widthChecker_sound
    (value : ⟦drawing⟧)
    (h : (widthChecker.run value).isEmpty = true)
    : WidthsFit value := by
    ..
  #+end_src

* RefinedDataType: checker design
#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
structure CheckedProperty (t : DataType) where
  property : ⟦t⟧ → Prop
  checker : Checker t
  sound : ∀ value, (checker.run value).isEmpty = true → property value

inductive RefinedDataType : (unrefined : DataType) → Type 1 where
  | refine {t : DataType}
      (refined : RefinedDataType t)
      (checked : CheckedProperty t) -- New
      : RefinedDataType t
  | base ..
  | sum ..
  | record ..
  | list ..
#+end_src

* Server-side decisions
What do we do when the property we want to verify cannot be expressed with a
=Checker=?

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
inductive ServerDecision {t : DataType}
    (property : ⟦t⟧ → Prop) (value : ⟦t⟧) where
  | valid (proof : property value)
  | invalid
      (proof : ¬ property value)
      (message : String)
      (targets : ValidTargets t value)

abbrev DecidableServerPred {t : DataType}
    (property : ⟦t⟧ → Prop) :=
  (value : ⟦t⟧) → ServerDecision property value
#+end_src

* Server-side refinements
Server-side refinements are checked only on submission.

#+attr_latex: :options basicstyle=\ttfamily\scriptsize
#+begin_src lean
structure ServerProperty (t : DataType) where
  property : ⟦t⟧ → Prop
  check : DecidableServerPred property
#+end_src
Advantages:
1. We validate in Lean, we don't need a serializable =Checker=.
2. Type safe error locations.

Disadvantages:
1. No immediate feedback on browser.

* Summary
1. Refinements add constraints to data types.
2. Validated values have proofs of those constraints.
3. Serializable checkers run in both the browser and the server.
4. Server-only checks use decidable Lean predicates and type safe error locations.
5. We can write specifications for checkers and (let the llm) prove that they
   are respected.
